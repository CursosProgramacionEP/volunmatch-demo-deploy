import { useState } from "react";
import { ArrowLeft, Plus, Send, X, Calendar } from "lucide-react";
import DatePicker from "react-datepicker";
import "react-datepicker/dist/react-datepicker.css";
import { es } from "date-fns/locale";
import { Button } from "../../components/comunes/Boton.jsx";
import { Field } from "../../components/comunes/Campo.jsx";
import { api } from "../../services/cliente-api.js";
import { go } from "../../navigation/navegacion.js";
import { categories } from "../../constants/aplicacion.js";

function emptyTask() {
  return { title: "", description: "", area: "General", priority: "Media" };
}

// Convierte un Date a string ISO para enviar al backend
function toISOString(date) {
  if (!date) return "";
  const pad = (n) => String(n).padStart(2, "0");
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}`;
}

/**
 * Valida y publica una iniciativa, mostrando errores directamente en cada campo.
 * @param {{token: string, showNotice: Function}} props Sesión de empresa y emisor de avisos.
 * @returns {JSX.Element} Formulario de creación de iniciativa.
 */
export function NewInitiativePage({ token, showNotice }) {
  const [form, setForm] = useState({
    title: "",
    category: "Medio ambiente",
    summary: "",
    date: "",
    endDate: "",
    location: "",
    quota: "",
    requirements: ["Mayor de 18 años"],
    tasks: [emptyTask()],
  });
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [errors, setErrors] = useState({});

  const update = (event) => {
    const { name, value } = event.target;
    setForm((current) => ({
      ...current,
      [name]: value,
    }));
    setErrors((current) => ({ ...current, [name]: "" }));
    setError("");
  };

  const updateTask = (index, field, value) =>
    setForm((current) => ({
      ...current,
      tasks: current.tasks.map((task, taskIndex) =>
        taskIndex === index ? { ...task, [field]: value } : task,
      ),
    }));

  const updateRequirement = (index, value) =>
    setForm((current) => ({
      ...current,
      requirements: current.requirements.map((item, itemIndex) =>
        itemIndex === index ? value : item,
      ),
    }));

  const addRequirement = () =>
    setForm((current) => ({
      ...current,
      requirements: [...current.requirements, ""],
    }));

  const removeRequirement = (index) =>
    setForm((current) => ({
      ...current,
      requirements: current.requirements.filter(
        (_item, itemIndex) => itemIndex !== index,
      ),
    }));

  const addTask = () =>
    setForm((current) => ({
      ...current,
      tasks: [...current.tasks, emptyTask()],
    }));

  const removeTask = (index) =>
    setForm((current) => ({
      ...current,
      tasks: current.tasks.filter((_task, taskIndex) => taskIndex !== index),
    }));

  function validate() {
    const nextErrors = {};
    const title = form.title.trim();
    const onlyLetters = /^[A-Za-zÁÉÍÓÚÜÑáéíóúüñ ]{3,250}$/;
    if (!title) nextErrors.title = "El nombre de la iniciativa es obligatorio.";
    else if (!onlyLetters.test(title)) nextErrors.title = "Usa al menos 3 letras, sin números ni símbolos.";
    if (!form.summary.trim()) nextErrors.summary = "Describe el objetivo de la iniciativa.";
    if (!form.location.trim()) nextErrors.location = "Indica el lugar de la actividad.";
    if (!form.quota) nextErrors.quota = "Indica el cupo de voluntarios.";
    else if (!Number.isInteger(Number(form.quota)) || Number(form.quota) <= 0) nextErrors.quota = "El cupo debe ser un número entero mayor que cero.";
    if (!form.date) nextErrors.date = "Selecciona la fecha de inicio.";
    if (!form.endDate) nextErrors.endDate = "Selecciona la fecha de fin.";
    if (form.date && form.endDate && form.endDate < form.date) nextErrors.endDate = "La fecha de fin debe ser igual o posterior a la de inicio.";
    setErrors(nextErrors);
    return Object.keys(nextErrors).length === 0;
  }

  async function submit(event) {
    event.preventDefault();
    if (!validate()) {
      setError("Revisa los campos marcados antes de publicar la iniciativa.");
      return;
    }
    setBusy(true);
    setError("");
    try {
      // Convertir fechas a string ISO antes de enviar
      const payload = {
        ...form,
        date: toISOString(form.date),
        endDate: toISOString(form.endDate),
      };
      const response = await api.createInitiative(token, payload);
      showNotice({ text: "Iniciativa publicada correctamente." });
      go(`/initiative/${response.initiative.id}`);
    } catch (requestError) {
      setError(requestError.message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <section className="workspace-content new-initiative">
      <button className="back-link" onClick={() => go("/company/dashboard")}>
        <ArrowLeft size={18} /> Volver a mis iniciativas
      </button>

      <div className="form-page-head">
        <p className="section-kicker">NUEVA CONVOCATORIA</p>
        <h1>Publica una iniciativa</h1>
        <p>
          Completa la información esencial y añade tareas iniciales para
          organizar el trabajo desde el primer día.
        </p>
      </div>

      <form className="initiative-form" onSubmit={submit} noValidate>
        <section>
          <h2>Información de la iniciativa</h2>
          <Field
            label="Nombre de la iniciativa"
            name="title"
            placeholder="Ej. Jornada de limpieza comunitaria"
            value={form.title}
            onChange={update}
            error={errors.title}
            required
          />
          <div className="form-grid">
            <Field
              label="Categoría"
              name="category"
              as="select"
              value={form.category}
              onChange={update}
              options={categories.slice(1)}
              required
            />
            <Field
              label="Cupo de voluntarios"
              name="quota"
              type="number"
              min="1"
              placeholder="Ej. 25"
              value={form.quota}
              onChange={update}
              error={errors.quota}
              required
            />
          </div>
          <Field
            label="Objetivo y descripción"
            name="summary"
            as="textarea"
            placeholder="Describe el propósito de la iniciativa y lo que se espera de los voluntarios."
            value={form.summary}
            onChange={update}
            error={errors.summary}
            required
          />

          <h2 style={{ marginTop: 24, marginBottom: 4 }}>Fecha</h2>
          <p style={{ color: "#666", fontSize: 14, marginBottom: 16 }}>
            Indica el inicio y fin de la iniciativa.
          </p>

          <div className="form-grid">
            <label className={`field ${errors.date ? "field--invalid" : ""}`}>
              <span>Inicio de la iniciativa <em aria-hidden="true">*</em></span>
              <div className="datepicker-wrapper">
                <Calendar size={16} className="datepicker-icon" />
                <DatePicker
                  selected={form.date ? new Date(form.date) : null}
                  onChange={(date) => {
                    setForm((current) => ({ ...current, date }));
                    setErrors((current) => ({ ...current, date: "" }));
                  }}
                  dateFormat="dd/MM/yyyy"
                  locale={es}
                  placeholderText="Selecciona la fecha"
                  className={`datepicker-input ${errors.date ? "input-error" : ""}`}
                  required
                />
              </div>
              {errors.date && <small className="field-error" role="alert">{errors.date}</small>}
            </label>
            <label className={`field ${errors.endDate ? "field--invalid" : ""}`}>
              <span>Fin de la iniciativa <em aria-hidden="true">*</em></span>
              <div className="datepicker-wrapper">
                <Calendar size={16} className="datepicker-icon" />
                <DatePicker
                  selected={form.endDate ? new Date(form.endDate) : null}
                  onChange={(date) => {
                    setForm((current) => ({ ...current, endDate: date }));
                    setErrors((current) => ({ ...current, endDate: "" }));
                  }}
                  dateFormat="dd/MM/yyyy"
                  locale={es}
                  placeholderText="Selecciona la fecha"
                  className={`datepicker-input ${errors.endDate ? "input-error" : ""}`}
                  required
                />
              </div>
              {errors.endDate && <small className="field-error" role="alert">{errors.endDate}</small>}
            </label>
          </div>

          <Field
            label="Lugar"
            name="location"
            placeholder="Ej. Parque Zonal, distrito"
            value={form.location}
            onChange={update}
            error={errors.location}
            required
          />
        </section>

        <section>
          <div className="form-section-head">
            <div>
              <h2>Requisitos</h2>
              <p>Indica lo que debe conocer o llevar el voluntario.</p>
            </div>
            <button type="button" className="add-link" onClick={addRequirement}>
              <Plus size={16} />
              Agregar requisito
            </button>
          </div>
          {form.requirements.map((requirement, index) => (
            <div className="editable-row" key={`requirement-${index}`}>
              <input
                value={requirement}
                onChange={(event) =>
                  updateRequirement(index, event.target.value)
                }
                placeholder="Ej. Disponibilidad de 4 horas"
              />
              <button
                type="button"
                aria-label="Eliminar requisito"
                onClick={() => removeRequirement(index)}
              >
                <X size={16} />
              </button>
            </div>
          ))}
        </section>

        <section>
          <div className="form-section-head">
            <div>
              <h2>Tareas iniciales</h2>
              <p>
                Los voluntarios aceptados podrán tomarlas y moverlas en el
                tablero.
              </p>
            </div>
            <button type="button" className="add-link" onClick={addTask}>
              <Plus size={16} />
              Agregar tarea
            </button>
          </div>
          {form.tasks.map((task, index) => (
            <div className="task-editor" key={`task-${index}`}>
              <div className="task-editor__head">
                <span>Tarea {index + 1}</span>
                {form.tasks.length > 1 && (
                  <button type="button" onClick={() => removeTask(index)}>
                    <X size={16} />
                    Quitar
                  </button>
                )}
              </div>
              <div className="form-grid">
                <Field
                  label="Nombre"
                  value={task.title}
                  onChange={(event) =>
                    updateTask(index, "title", event.target.value)
                  }
                  placeholder="Nombre de la tarea"
                />
                <Field
                  label="Área"
                  value={task.area}
                  onChange={(event) =>
                    updateTask(index, "area", event.target.value)
                  }
                  placeholder="Ej. Logística"
                />
              </div>
              <Field
                label="Descripción"
                as="textarea"
                value={task.description}
                onChange={(event) =>
                  updateTask(index, "description", event.target.value)
                }
                placeholder="¿Qué debe hacer el voluntario?"
              />
              <Field
                label="Prioridad"
                as="select"
                value={task.priority}
                onChange={(event) =>
                  updateTask(index, "priority", event.target.value)
                }
                options={["Alta", "Media", "Baja"]}
              />
            </div>
          ))}
        </section>

        {error && <p className="form-error">{error}</p>}

        <footer className="form-footer">
          <Button
            variant="secondary"
            type="button"
            onClick={() => go("/company/dashboard")}
          >
            Cancelar
          </Button>
          <Button type="submit" icon={Send} disabled={busy}>
            {busy ? "Publicando..." : "Publicar iniciativa"}
          </Button>
        </footer>
      </form>
    </section>
  );
}
