import { useEffect, useState } from 'react';
import { ArrowLeft, Ellipsis, Sparkles, Users, Plus } from 'lucide-react';
import { Loader } from '../../components/comunes/Cargador.jsx';
import { EmptyState } from '../../components/comunes/EstadoVacio.jsx';
import { TaskCard } from '../../components/tablero/TarjetaTarea.jsx';
import { ModalNuevaTarea } from './ModalNuevaTarea.jsx';
import { Modal } from '../../components/comunes/VentanaModal.jsx';
import { Button } from '../../components/comunes/Boton.jsx';
import { api } from '../../services/cliente-api.js';
import { go } from '../../navigation/navegacion.js';
import { statusMeta } from '../../constants/aplicacion.js';

const columns = ['pending', 'in_progress', 'done'];

export function KanbanPage({ token, user, initiativeId, showNotice }) {
  const [initiative, setInitiative] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [draggedId, setDraggedId] = useState(null);
  const [showNewTask, setShowNewTask] = useState(false);
  const [pendingMove, setPendingMove] = useState(null);

  const isCompany = user?.role === 'company';

  const load = () => {
    setLoading(true);
    api.board(token, initiativeId)
      .then((response) => setInitiative(response.initiative))
      .catch((requestError) => setError(requestError.message))
      .finally(() => setLoading(false));
  };
  useEffect(load, [token, initiativeId]);

  /** Actualiza una tarea y comunica al usuario el resultado de una acción confirmada. */
  async function updateTask(taskId, payload, successText = '') {
    try {
      await api.updateTask(token, initiative.id, taskId, payload);
      load();
      if (successText) showNotice({ text: successText });
      return true;
    } catch (requestError) {
      showNotice({ type: 'error', text: requestError.message });
      return false;
    }
  }

  async function onDrop(status) {
    if (!draggedId) return;

    // Bloquear arrastre si la tarea ya está finalizada
    const dragged = initiative.tasks.find((t) => t.id === draggedId);
    if (dragged?.status === 'done' && status !== 'done') {
      showNotice({ type: 'error', text: 'No puedes mover una tarea finalizada.' });
      setDraggedId(null);
      return;
    }

    if (status === 'done' && dragged.status !== 'done') {
      setPendingMove({ task: dragged, status });
    } else {
      await updateTask(draggedId, { status }, 'El estado de la tarea fue actualizado.');
    }
    setDraggedId(null);
  }

  /** Evita finalizar una tarea por accidente al requerir una confirmación explícita. */
  async function confirmPendingMove() {
    if (!pendingMove) return;
    const updated = await updateTask(pendingMove.task.id, { status: pendingMove.status }, 'Tarea marcada como finalizada.');
    if (updated) setPendingMove(null);
  }

  async function handleCrearTarea(formData) {
    await api.createTask(token, initiative.id, formData);
    setShowNewTask(false);
    await load();
    showNotice({ type: 'success', text: 'Tarea creada correctamente.' });
  }

  if (loading) return <section className="workspace-content"><Loader label="Cargando tablero..." /></section>;

  if (error || !initiative) {
    return (
      <section className="workspace-content">
        <EmptyState
          title="No puedes abrir este tablero"
          text={error || 'Necesitas tener una postulación aceptada.'}
          action={() => go(user.role === 'company' ? '/company/dashboard' : '/volunteer/activities')}
          actionLabel="Volver"
        />
      </section>
    );
  }

  return (
    <section className="kanban-page">
      <header className="kanban-top">
        <button className="back-link" onClick={() => go(user.role === 'company' ? `/initiative/${initiative.id}` : '/volunteer/activities')}>
          <ArrowLeft size={18} /> Volver
        </button>
        <div>
          <p className="section-kicker">{initiative.category}</p>
          <h1>{initiative.title}</h1>
        </div>
        <span className="board-role">
          <Users size={17} />{isCompany ? 'Vista de empresa' : 'Voluntario'}
        </span>
      </header>

      <div className="kanban-guide">
        <Sparkles size={17} />
        {isCompany
          ? 'Sigue el avance de las tareas asignadas a tu iniciativa.'
          : 'Arrastra tus tareas entre columnas para actualizar tu avance.'}
      </div>

      {/* 🆕 Botón "+ Agregar tarea" — solo visible para empresa */}
      {isCompany && (
        <div className="kanban-actions">
          <button
            type="button"
            className="btn-add-task"
            onClick={() => setShowNewTask(true)}
          >
            <Plus size={16} /> Agregar tarea
          </button>
        </div>
      )}

      <div className="kanban-board">
        {columns.map((column) => {
          const tasks = initiative.tasks.filter((task) => task.status === column);
          return (
            <section
              className="kanban-column"
              key={column}
              onDragOver={(event) => event.preventDefault()}
              onDrop={() => onDrop(column)}
            >
              <header>
                <div>
                  <h2>{statusMeta[column].label}</h2>
                  <span>{tasks.length}</span>
                </div>
                <button aria-label="Más opciones"><Ellipsis size={20} /></button>
              </header>
              <div className="kanban-column__cards">
                {tasks.map((task) => (
                  <TaskCard
                    key={task.id}
                    task={task}
                    currentUser={user}
                    onDragStart={() => setDraggedId(task.id)}
                    onClaim={() => updateTask(task.id, { claim: true }, 'Tomaste la tarea. Ya puedes registrar tu avance.')}
                    onFinish={() => setPendingMove({ task, finish: true })}
                  />
                ))}
                {!tasks.length && <div className="drop-hint">Suelta una tarea aquí</div>}
              </div>
            </section>
          );
        })}
      </div>

      {/* 🆕 Modal para agregar tarea — solo empresa */}
      {showNewTask && (
        <ModalNuevaTarea
          onCrear={handleCrearTarea}
          onCancelar={() => setShowNewTask(false)}
        />
      )}
      {pendingMove && (
        <ConfirmFinishModal
          task={pendingMove.task}
          onCancel={() => setPendingMove(null)}
          onConfirm={async () => {
            if (pendingMove.finish) {
              const updated = await updateTask(pendingMove.task.id, { finish: true }, 'Tarea marcada como finalizada.');
              if (updated) setPendingMove(null);
              return;
            }
            await confirmPendingMove();
          }}
        />
      )}
    </section>
  );
}

function ConfirmFinishModal({ task, onCancel, onConfirm }) {
  return (
    <Modal onClose={onCancel}>
      <div className="confirm-modal">
        <span className="confirm-modal__icon"><Sparkles size={30} /></span>
        <h2>¿Finalizar esta tarea?</h2>
        <p>Confirmarás que <b>{task.title}</b> está terminada. Después no podrá volver a una columna anterior.</p>
        <div className="confirm-modal__actions"><Button variant="secondary" onClick={onCancel}>Volver</Button><Button onClick={onConfirm}>Sí, finalizar</Button></div>
      </div>
    </Modal>
  );
}
