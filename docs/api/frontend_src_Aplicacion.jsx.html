import { useEffect, useMemo, useState } from 'react';
import { Notice } from './components/comunes/Aviso.jsx';
import { AppLayout } from './components/aplicacion/EstructuraAplicacion.jsx';
import { HomePage } from './pages/inicio/PaginaInicio.jsx';
import { AuthPage } from './pages/autenticacion/PaginaAutenticacion.jsx';
import { ExplorePage } from './pages/exploracion/PaginaExploracion.jsx';
import { InitiativeDetailPage } from './pages/iniciativas/PaginaDetalleIniciativa.jsx';
import { ActivitiesPage } from './pages/actividades/PaginaActividades.jsx';
import { CompanyDashboard } from './pages/empresa/PanelEmpresa.jsx';
import { NewInitiativePage } from './pages/iniciativas/PaginaNuevaIniciativa.jsx';
import { KanbanPage } from './pages/tablero/PaginaTablero.jsx';
import { ProfilePage, CompanyPublicProfilePage } from './pages/perfil/PaginaPerfil.jsx';
import { HistoryPage } from './pages/historial/PaginaHistorial.jsx';
import { NotificationsPage } from './pages/notificaciones/PaginaNotificaciones.jsx';
import { go, useRoute } from './navigation/navegacion.js';

// Componente raíz de la aplicación.
// Se encarga de tres cosas:
//   1) Guardar la sesión (usuario + token) en localStorage.
//   2) Decidir, según la ruta actual y si hay sesión, qué pantalla mostrar.
//   3) Mostrar el aviso flotante (Notice) por encima de cualquier pantalla.
export default function App() {
  const route = useRoute();
  const [session, setSession] = useState(() => {
    try {
      return JSON.parse(localStorage.getItem('volunmatch-session') || 'null');
    } catch {
      return null;
    }
  });
  const [notice, setNotice] = useState(null);

  const user = session?.user;
  const token = session?.token;
  const showNotice = (item) => setNotice(item);

  // Extrae el id de la iniciativa desde rutas como "/initiative/123" o "/kanban/123".
  const routeParts = useMemo(() => route.split('/').filter(Boolean), [route]);
  const initiativeId = routeParts[0] === 'initiative' || routeParts[0] === 'kanban' ? routeParts[1] : null;
  const companyId = routeParts[0] === 'company' && routeParts[1] === 'profile' ? routeParts[2] : null;

  // Si no hay sesión y el usuario intenta ver una pantalla protegida, lo mandamos a /auth.
  useEffect(() => {
    if (!user && route !== '/' && route !== '/auth') go('/auth');
  }, [route, user]);

  function onAuthenticated(nextSession) {
    localStorage.setItem('volunmatch-session', JSON.stringify(nextSession));
    setSession(nextSession);
  }

  /** Actualiza el usuario guardado cuando el perfil se modifica correctamente. */
  function onSessionUpdated(profile) {
    setSession((current) => {
      const nextUser = {
        ...current.user,
        fullName: profile.role === 'company' ? profile.companyName : profile.fullName,
        email: profile.email,
      };
      const nextSession = { ...current, user: nextUser };
      localStorage.setItem('volunmatch-session', JSON.stringify(nextSession));
      return nextSession;
    });
  }

  let content;

  if (!user) {
    // Sin sesión: solo se puede ver el inicio o la pantalla de acceso.
    content = route === '/' ? <HomePage user={null} /> : <AuthPage onAuthenticated={onAuthenticated} showNotice={showNotice} />;
  } else if (route === '/') {
    content = <HomePage user={user} />;
  } else {
    // Con sesión: se elige la pantalla según la ruta y el rol del usuario.
    let page;
    if (route.startsWith('/initiative/')) {
      page = <InitiativeDetailPage token={token} user={user} initiativeId={initiativeId} showNotice={showNotice} />;
    } else if (route.startsWith('/kanban/')) {
      page = <KanbanPage token={token} user={user} initiativeId={initiativeId} showNotice={showNotice} />;
    } else if (route === '/profile') {
      page = <ProfilePage token={token} user={user} showNotice={showNotice} onSessionUpdated={onSessionUpdated} />;
    } else if (companyId && user.role === 'volunteer') {
      page = <CompanyPublicProfilePage token={token} companyId={companyId} />;
    } else if (route === '/notifications') {
      page = <NotificationsPage token={token} showNotice={showNotice} />;
    } else if (route === '/company/history' || route === '/volunteer/history') {
      page = <HistoryPage token={token} user={user} />;
    } else if (user.role === 'company' && route === '/company/new') {
      page = <NewInitiativePage token={token} showNotice={showNotice} />;
    } else if (user.role === 'company') {
      page = <CompanyDashboard token={token} showNotice={showNotice} />;
    } else if (route === '/volunteer/activities') {
      page = <ActivitiesPage token={token} />;
    } else {
      page = (
        <ExplorePage
          token={token}
          onOpen={(initiative) => go(initiative.isMember || initiative.applicationStatus === 'accepted' ? `/kanban/${initiative.id}` : `/initiative/${initiative.id}`)}
          showNotice={showNotice}
        />
      );
    }
    content = <AppLayout user={user} route={route} token={token}>{page}</AppLayout>;
  }

  return (
    <>
      {content}
      <Notice notice={notice} onClose={() => setNotice(null)} />
    </>
  );
}
