import { useEffect, useState } from 'react';
import { Award, Building2, CalendarDays, CheckCircle2, Edit3, Mail, Phone, Plus, Save, UserRound, Users } from 'lucide-react';
import { Loader } from '../../components/comunes/Cargador.jsx';
import { EmptyState } from '../../components/comunes/EstadoVacio.jsx';
import { Button } from '../../components/comunes/Boton.jsx';
import { Field } from '../../components/comunes/Campo.jsx';
import { api } from '../../services/cliente-api.js';
import { go } from '../../navigation/navegacion.js';
import { categories } from '../../constants/aplicacion.js';

const blankCertification = { title: '', issuer: '', issuedAt: '', documentUrl: '' };

/**
 * Muestra y permite actualizar el perfil del usuario que inició sesión.
 * @param {object} props Propiedades de sesión, avisos y actualización local de usuario.
 * @returns {JSX.Element} Perfil personal o institucional.
 */
export function ProfilePage({ token, user, showNotice, onSessionUpdated }) {
  const [profile, setProfile] = useState(null);
  const [form, setForm] = useState(null);
  const [loading, setLoading] = useState(true);
  const [editing, setEditing] = useState(false);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');
  const [addingCertification, setAddingCertification] = useState(false);
  const [certification, setCertification] = useState(blankCertification);

  useEffect(() => {
    let active = true;
    api.myProfile(token)
      .then(({ profile: nextProfile }) => {
        if (!active) return;
        setProfile(nextProfile);
        setForm(profileToForm(nextProfile));
      })
      .catch((requestError) => active && setError(requestError.message))
      .finally(() => active && setLoading(false));
    return () => { active = false; };
  }, [token]);

  function updateForm(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
    setError('');
  }

  async function saveProfile(event) {
    event.preventDefault();
    setSaving(true);
    setError('');
    try {
      const response = await api.updateMyProfile(token, form);
      setProfile(response.profile);
      setForm(profileToForm(response.profile));
      setEditing(false);
      onSessionUpdated(response.profile);
      showNotice({ text: 'Cambios guardados correctamente.' });
    } catch (requestError) {
      setError(requestError.message);
    } finally {
      setSaving(false);
    }
  }

  async function saveCertification(event) {
    event.preventDefault();
    setSaving(true);
    setError('');
    try {
      const response = await api.addCertification(token, certification);
      setProfile((current) => ({ ...current, certifications: [response.certification, ...current.certifications] }));
      setCertification(blankCertification);
      setAddingCertification(false);
      showNotice({ text: 'Certificación agregada correctamente.' });
    } catch (requestError) {
      setError(requestError.message);
    } finally {
      setSaving(false);
    }
  }

  if (loading) return <section className="workspace-content"><Loader label="Cargando perfil..." /></section>;
  if (error && !profile) return <section className="workspace-content"><EmptyState title="No pudimos cargar tu perfil" text={error} /></section>;

  return (
    <section className="workspace-content profile-page">
      <ProfileHeader profile={profile} editable onEdit={() => setEditing((open) => !open)} />
      {editing ? (
        <ProfileForm
          profile={profile}
          form={form}
          error={error}
          saving={saving}
          onChange={updateForm}
          onCancel={() => { setEditing(false); setForm(profileToForm(profile)); setError(''); }}
          onSubmit={saveProfile}
        />
      ) : (
        <ProfileContent
          profile={profile}
          showCertifications={profile.role === 'volunteer'}
          addingCertification={addingCertification}
          certification={certification}
          error={error}
          saving={saving}
          onAddCertification={() => setAddingCertification(true)}
          onCancelCertification={() => { setAddingCertification(false); setCertification(blankCertification); setError(''); }}
          onCertificationChange={(event) => setCertification((current) => ({ ...current, [event.target.name]: event.target.value }))}
          onCertificationSubmit={saveCertification}
        />
      )}
    </section>
  );
}

/**
 * Presenta a los voluntarios el perfil institucional de una empresa y sus iniciativas activas.
 * @param {{token: string, companyId: string}} props Datos de sesión e identificador de empresa.
 * @returns {JSX.Element} Perfil institucional de consulta.
 */
export function CompanyPublicProfilePage({ token, companyId }) {
  const [profile, setProfile] = useState(null);
  const [error, setError] = useState('');

  useEffect(() => {
    api.companyProfile(token, companyId)
      .then(({ profile: nextProfile }) => setProfile(nextProfile))
      .catch((requestError) => setError(requestError.message));
  }, [token, companyId]);

  if (!profile && !error) return <section className="workspace-content"><Loader label="Cargando perfil de empresa..." /></section>;
  if (error) return <section className="workspace-content"><EmptyState title="No pudimos abrir este perfil" text={error} action={() => go('/volunteer/explore')} actionLabel="Volver a explorar" /></section>;

  return (
    <section className="workspace-content profile-page">
      <ProfileHeader profile={profile} />
      <ProfileContent profile={profile} />
    </section>
  );
}

/** @returns {object} Valores editables derivados del perfil recibido por la API. */
function profileToForm(profile) {
  return profile.role === 'volunteer'
    ? { fullName: profile.fullName, email: profile.email, phone: profile.phone, bio: profile.bio }
    : { companyName: profile.companyName, contactName: profile.contactName, email: profile.email, sector: profile.sector };
}

function ProfileHeader({ profile, editable = false, onEdit }) {
  const name = profile.role === 'company' ? profile.companyName : profile.fullName;
  return (
    <header className="profile-hero">
      <span className="profile-hero__avatar">{name.split(' ').map((part) => part[0]).slice(0, 2).join('')}</span>
      <div>
        <p className="section-kicker">{profile.role === 'company' ? 'PERFIL INSTITUCIONAL' : 'PERFIL DE VOLUNTARIO'}</p>
        <h1>{name}</h1>
        <p>{profile.role === 'company' ? `${profile.sector} · Organización verificada` : 'Tu información de contacto y trayectoria de voluntariado.'}</p>
      </div>
      {editable && <Button className="profile-hero__edit" icon={Edit3} onClick={onEdit}>Editar perfil</Button>}
    </header>
  );
}

function ProfileContent({ profile, showCertifications = false, addingCertification, certification, error, saving, onAddCertification, onCancelCertification, onCertificationChange, onCertificationSubmit }) {
  if (profile.role === 'company') return <CompanyProfileContent profile={profile} />;

  return (
    <div className="profile-layout">
      <section className="profile-card profile-card--main">
        <h2><UserRound size={20} /> Información de contacto</h2>
        <div className="profile-details">
          <span><Mail size={17} />{profile.email}</span>
          <span><Phone size={17} />{profile.phone || 'Teléfono sin registrar'}</span>
          <span><CalendarDays size={17} />Miembro desde {formatDate(profile.createdAt)}</span>
        </div>
        <h3>Sobre mí</h3>
        <p>{profile.bio || 'Aún no has agregado una descripción personal.'}</p>
      </section>
      <section className="profile-stats">
        <Metric icon={Users} value={profile.acceptedInitiatives} label="Iniciativas aceptadas" />
        <Metric icon={CheckCircle2} value={profile.completedTasks} label="Tareas completadas" />
      </section>
      {showCertifications && (
        <section className="profile-card profile-card--main certifications-card">
          <div className="profile-section-head">
            <div><h2><Award size={20} /> Certificaciones</h2><p>Registra los certificados que acreditan tu participación.</p></div>
            <Button variant="secondary" icon={Plus} onClick={onAddCertification}>Agregar</Button>
          </div>
          {addingCertification && <CertificationForm certification={certification} error={error} saving={saving} onChange={onCertificationChange} onCancel={onCancelCertification} onSubmit={onCertificationSubmit} />}
          {profile.certifications.length ? (
            <div className="certification-list">
              {profile.certifications.map((item) => <CertificationItem key={item.id} item={item} />)}
            </div>
          ) : !addingCertification && <p className="profile-empty">Aún no registraste certificaciones.</p>}
        </section>
      )}
    </div>
  );
}

function CompanyProfileContent({ profile }) {
  return (
    <div className="profile-layout">
      <section className="profile-card profile-card--main">
        <h2><Building2 size={20} /> Datos de la organización</h2>
        <div className="profile-details">
          <span><Mail size={17} />{profile.email}</span>
          <span><UserRound size={17} />Responsable: {profile.contactName}</span>
          <span><CheckCircle2 size={17} />RUC: {profile.ruc}</span>
        </div>
      </section>
      <section className="profile-card profile-card--main company-profile-initiatives">
        <div className="profile-section-head"><div><h2><Users size={20} /> Iniciativas publicadas</h2><p>Convocatorias vinculadas a esta organización.</p></div></div>
        {profile.initiatives.length ? (
          <div className="profile-initiative-list">
            {profile.initiatives.map((initiative) => (
              <article key={initiative.id}>
                <span className="category-chip">{initiative.category}</span>
                <div><h3>{initiative.title}</h3><p>{initiative.summary}</p><small><CalendarDays size={14} /> {formatDate(initiative.date)} · {initiative.location}</small></div>
                <Button variant="secondary" onClick={() => go(`/initiative/${initiative.id}`)}>Ver iniciativa</Button>
              </article>
            ))}
          </div>
        ) : <p className="profile-empty">No hay iniciativas activas para mostrar.</p>}
      </section>
    </div>
  );
}

function ProfileForm({ profile, form, error, saving, onChange, onCancel, onSubmit }) {
  const volunteer = profile.role === 'volunteer';
  return (
    <form className="profile-card profile-form" onSubmit={onSubmit} noValidate>
      <h2>{volunteer ? 'Editar datos personales' : 'Editar datos de empresa'}</h2>
      {volunteer ? (
        <>
          <Field label="Nombre y apellidos" name="fullName" value={form.fullName} onChange={onChange} required />
          <div className="form-grid"><Field label="Correo electrónico" name="email" type="email" value={form.email} onChange={onChange} required /><Field label="Teléfono" name="phone" value={form.phone} onChange={onChange} /></div>
          <Field label="Biografía" name="bio" as="textarea" value={form.bio} onChange={onChange} />
        </>
      ) : (
        <>
          <div className="form-grid"><Field label="Nombre de empresa" name="companyName" value={form.companyName} onChange={onChange} required /><Field label="Responsable" name="contactName" value={form.contactName} onChange={onChange} required /></div>
          <div className="form-grid"><Field label="Correo corporativo" name="email" type="email" value={form.email} onChange={onChange} required /><Field label="Sector" name="sector" as="select" value={form.sector} onChange={onChange} options={categories.slice(1).concat('Otro')} required /></div>
        </>
      )}
      {error && <p className="form-error" role="alert">{error}</p>}
      <footer className="form-footer"><Button type="button" variant="secondary" onClick={onCancel}>Cancelar</Button><Button type="submit" icon={Save} disabled={saving}>{saving ? 'Guardando...' : 'Guardar cambios'}</Button></footer>
    </form>
  );
}

function CertificationForm({ certification, error, saving, onChange, onCancel, onSubmit }) {
  return (
    <form className="certification-form" onSubmit={onSubmit} noValidate>
      <div className="form-grid"><Field label="Nombre de la certificación" name="title" value={certification.title} onChange={onChange} required /><Field label="Entidad emisora" name="issuer" value={certification.issuer} onChange={onChange} required /></div>
      <div className="form-grid"><Field label="Fecha de emisión" name="issuedAt" type="date" value={certification.issuedAt} onChange={onChange} /><Field label="Enlace al documento" name="documentUrl" type="url" placeholder="https://..." value={certification.documentUrl} onChange={onChange} /></div>
      {error && <p className="form-error" role="alert">{error}</p>}
      <footer className="form-footer"><Button type="button" variant="secondary" onClick={onCancel}>Cancelar</Button><Button type="submit" disabled={saving}>{saving ? 'Guardando...' : 'Guardar certificación'}</Button></footer>
    </form>
  );
}

function CertificationItem({ item }) {
  return <article><span className="certification-icon"><Award size={18} /></span><div><b>{item.title}</b><small>{item.issuer} · {formatDate(item.issuedAt)}</small></div>{item.documentUrl && <a href={item.documentUrl} target="_blank" rel="noreferrer">Ver documento</a>}</article>;
}

function Metric({ icon: Icon, value, label }) {
  return <article><span><Icon size={20} /></span><div><b>{value}</b><small>{label}</small></div></article>;
}

function formatDate(value) {
  if (!value) return 'Fecha por definir';
  return new Intl.DateTimeFormat('es-PE', { day: '2-digit', month: 'short', year: 'numeric' }).format(new Date(value));
}
