import { useState } from 'react';
import { ArrowLeft, ArrowRight, BriefcaseBusiness, HeartHandshake, Leaf } from 'lucide-react';
import { Brand } from '../../components/autenticacion/Marca.jsx';
import { Button } from '../../components/comunes/Boton.jsx';
import { Field } from '../../components/comunes/Campo.jsx';
import { api } from '../../services/cliente-api.js';
import { go } from '../../navigation/navegacion.js';
import { categories } from '../../constants/aplicacion.js';

const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

/**
 * Gestiona acceso, registro y solicitud de recuperación de contraseña.
 * @param {{onAuthenticated: Function, showNotice: Function}} props Acciones que actualizan la sesión y los avisos.
 * @returns {JSX.Element} Pantalla de autenticación de VolunMatch.
 */
export function AuthPage({ onAuthenticated, showNotice }) {
  const [mode, setMode] = useState('login');
  const [role, setRole] = useState('volunteer');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');
  const [emailError, setEmailError] = useState('');
  const [recoverySent, setRecoverySent] = useState(false);
  const [form, setForm] = useState({
    fullName: '', email: '', password: '', phone: '', bio: '',
    companyName: '', ruc: '', sector: 'Medio ambiente', contactName: '',
  });

  const isRegister = mode === 'register';
  const isRecovery = mode === 'recovery';

  function update(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
    if (name === 'email') setEmailError('');
    setError('');
  }

  function validateEmail() {
    const nextError = emailPattern.test(form.email.trim()) ? '' : 'Ingresa un correo electrónico válido.';
    setEmailError(nextError);
    return !nextError;
  }

  async function submit(event) {
    event.preventDefault();
    if (!validateRegistration()) return;
    if (!validateEmail()) return;
    setBusy(true);
    setError('');
    try {

      const finalFormData = {
        ...form,
        sector: form.sector === "Otro" ? form.customSector : form.sector,
      };
      
      const response = isRegister
        ? await api.register({ ...finalFormData, role })
        : await api.login({ email: form.email, password: form.password });
      onAuthenticated(response);
      showNotice({ text: isRegister ? 'Tu cuenta fue creada. ¡Bienvenido a VolunMatch!' : 'Sesión iniciada correctamente.' });
      go(response.user.role === 'company' ? '/company/dashboard' : '/volunteer/explore');
    } catch (requestError) {
      setError(!isRegister && requestError.message === 'Correo o contraseña incorrectos.'
        ? 'Correo o contraseña incorrectos. Revisa tus datos y verifica que elegiste el perfil correcto.'
        : requestError.message);
    } finally {
      setBusy(false);
    }
  }

  async function submitRecovery(event) {
    event.preventDefault();
    if (!validateEmail()) return;
    setBusy(true);
    setError('');
    try {
      const response = await api.recoverPassword({ email: form.email });
      setRecoverySent(true);
      showNotice({ text: response.message });
    } catch (requestError) {
      setError(requestError.message);
    } finally {
      setBusy(false);
    }
  }

  function changeMode(nextMode) {
    setMode(nextMode);
    setError('');
    setEmailError('');
    setRecoverySent(false);
  }

  return (
    <main className="auth-page">
      <div className="auth-page__top">
        <Brand />
        <button className="link-button" onClick={() => go('/')}>
          <ArrowLeft size={16} /> Volver al inicio
        </button>
      </div>

      <section className="auth-shell">
        <AuthAside />
        <div className="auth-panel">
          {isRecovery ? (
            <RecoveryForm
              form={form}
              busy={busy}
              error={error}
              emailError={emailError}
              recoverySent={recoverySent}
              onChange={update}
              onBlur={validateEmail}
              onSubmit={submitRecovery}
              onBack={() => changeMode('login')}
            />
          ) : (
            <>
              <div className="auth-panel__heading">
                <p className="section-kicker">{isRegister ? 'ÚNETE A LA RED' : 'BIENVENIDO DE NUEVO'}</p>
                <h2>{isRegister ? 'Crea tu cuenta' : 'Ingresa a VolunMatch'}</h2>
                <p>{isRegister ? 'Elige el perfil que representa tu participación.' : 'Accede para continuar con tus iniciativas y tareas.'}</p>
              </div>

              <div className="role-toggle" role="tablist" aria-label="Tipo de cuenta">
                <button type="button" className={role === 'volunteer' ? 'active' : ''} onClick={() => setRole('volunteer')}>
                  <HeartHandshake size={17} /> Voluntario
                </button>
                <button type="button" className={role === 'company' ? 'active' : ''} onClick={() => setRole('company')}>
                  <BriefcaseBusiness size={17} /> Empresa
                </button>
              </div>

              <form className="auth-form" onSubmit={submit} noValidate>
                {isRegister && role === 'volunteer' && (
                  <>
                    <Field label="Nombre y apellidos" name="fullName" placeholder="Ej. Ana García" value={form.fullName} onChange={update} required />
                    <div className="field-row">
                      <Field label="Teléfono" name="phone" placeholder="987 654 321" value={form.phone} onChange={update} />
                      <Field label="Correo electrónico" name="email" type="email" placeholder="ana@correo.com" value={form.email} onChange={update} onBlur={validateEmail} error={emailError} required />
                    </div>
                    <Field label="Cuéntanos brevemente de ti" name="bio" as="textarea" placeholder="Tus intereses o experiencia como voluntario..." value={form.bio} onChange={update} />
                  </>
                )}

                {isRegister && role === 'company' && (
                  <>
                    <div className="field-row">
                      <Field label="Razón social" name="companyName" placeholder="Nombre de la empresa" value={form.companyName} onChange={update} required />
                      <Field label="RUC" name="ruc" placeholder="20601234567" value={form.ruc} onChange={update} required />
                    </div>
                    <Field label="Nombre de contacto" name="contactName" placeholder="Persona responsable" value={form.contactName} onChange={update} />
                    <div className="field-row">
                      <Field label="Sector" name="sector" as="select" value={form.sector} onChange={update} options={categories.slice(1).concat('Otro')} required />
                      <Field label="Correo corporativo" name="email" type="email" placeholder="contacto@empresa.com" value={form.email} onChange={update} onBlur={validateEmail} error={emailError} required />
                    </div>
                  </>
                )}

                {!isRegister && (
                  <Field
                    label="Correo electrónico"
                    name="email"
                    type="email"
                    placeholder={role === 'volunteer' ? 'ana@volunmatch.pe' : 'empresa@volunmatch.pe'}
                    value={form.email}
                    onChange={update}
                    onBlur={validateEmail}
                    error={emailError}
                    required
                  />
                )}

                <Field
                  label="Contraseña"
                  name="password"
                  type="password"
                  placeholder={isRegister ? 'Crea una contraseña' : 'Tu contraseña'}
                  hint={isRegister ? 'Mínimo 6 caracteres.' : ''}
                  value={form.password}
                  onChange={update}
                  required
                />

                {!isRegister && (
                  <button type="button" className="forgot-password" onClick={() => changeMode('recovery')}>
                    ¿Olvidaste tu contraseña?
                  </button>
                )}
                {error && <p className="form-error" role="alert">{error}</p>}

                <Button type="submit" className="auth-form__submit" icon={ArrowRight} disabled={busy}>
                  {busy ? 'Procesando...' : (isRegister ? `Crear cuenta de ${role === 'volunteer' ? 'voluntario' : 'empresa'}` : 'Iniciar sesión')}
                </Button>
              </form>

              <p className="auth-switch">
                {isRegister ? '¿Ya tienes una cuenta?' : '¿Aún no tienes cuenta?'}{' '}
                <button type="button" onClick={() => changeMode(isRegister ? 'login' : 'register')}>
                  {isRegister ? 'Inicia sesión' : 'Regístrate'}
                </button>
              </p>
            </>
          )}
        </div>
      </section>
    </main>
  );
}

function AuthAside() {
  return (
    <aside className="auth-aside">
      <Brand inverse />
      <div className="auth-aside__content">
        <span className="eyebrow eyebrow--light"><Leaf size={14} /> ACCIONES QUE SUMAN</span>
        <h1>El cambio se organiza mejor en equipo.</h1>
        <p>Encuentra iniciativas, conecta con personas comprometidas y aporta desde tus habilidades.</p>
      </div>
      <div className="auth-aside__numbers">
        <div><b>+1,200</b><span>voluntarios</span></div>
        <div><b>86</b><span>iniciativas activas</span></div>
      </div>
      <div className="auth-aside__art">
        <span className="art-hill art-hill--one" />
        <span className="art-hill art-hill--two" />
        <span className="art-leaf art-leaf--one" />
        <span className="art-leaf art-leaf--two" />
      </div>
    </aside>
  );
}

function RecoveryForm({ form, busy, error, emailError, recoverySent, onChange, onBlur, onSubmit, onBack }) {
  return (
    <>
      <div className="auth-panel__heading">
        <p className="section-kicker">RECUPERAR ACCESO</p>
        <h2>¿Olvidaste tu contraseña?</h2>
        <p>Ingresa el correo registrado y te enviaremos las instrucciones para restablecerla.</p>
      </div>
      {recoverySent ? (
        <div className="recovery-success" role="status">
          <b>Revisa tu correo.</b>
          <span>Si la dirección está registrada, recibirás las instrucciones de recuperación.</span>
        </div>
      ) : (
        <form className="auth-form" onSubmit={onSubmit} noValidate>
          <Field label="Correo electrónico" name="email" type="email" placeholder="ejemplo@correo.com" value={form.email} onChange={onChange} onBlur={onBlur} error={emailError} required />
          {error && <p className="form-error" role="alert">{error}</p>}
          <Button type="submit" className="auth-form__submit" disabled={busy}>
            {busy ? 'Enviando...' : 'Enviar enlace de recuperación'}
          </Button>
        </form>
      )}
      <p className="auth-switch"><button type="button" onClick={onBack}><ArrowLeft size={15} /> Volver al inicio de sesión</button></p>
    </>
  );
}
