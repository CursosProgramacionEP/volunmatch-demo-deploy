import { useEffect, useState } from "react";
import {
  ArrowLeft,
  ArrowRight,
  Building2,
  CalendarDays,
  CheckCircle2,
  Clock3,
  HeartHandshake,
  Leaf,
  MapPin,
  ShieldCheck,
  Users,
  XCircle,
} from "lucide-react";
import { Loader } from "../../components/comunes/Cargador.jsx";
import { EmptyState } from "../../components/comunes/EstadoVacio.jsx";
import { Button } from "../../components/comunes/Boton.jsx";
import { Modal } from "../../components/comunes/VentanaModal.jsx";
import { api } from "../../services/cliente-api.js";
import { go } from "../../navigation/navegacion.js";

const initialForm = (user) => ({
  fullName: user.fullName || "",
  email: user.email || "",
  phone: "",
  schedule: "Mañana (08:00 AM - 12:00 PM)",
  saturday: true,
  sunday: false,
  motivation: "",
  experience: "",
  shirtSize: "M",
  restrictions: "",
  acceptedTerms: false,
});

export function InitiativeDetailPage({
  token,
  user,
  initiativeId,
  showNotice,
}) {
  const [initiative, setInitiative] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [step, setStep] = useState("detail");
  const [form, setForm] = useState(() => initialForm(user));
  const [busy, setBusy] = useState(false);
  const [yaPostulado, setYaPostulado] = useState(false);
  const [cancelOpen, setCancelOpen] = useState(false);
  const [cancelling, setCancelling] = useState(false);

  useEffect(() => {
    setLoading(true);
    Promise.all([
      api.initiative(token, initiativeId),
      api.verificarPostulacion(token, initiativeId).catch(() => ({ yaPostulado: false }))
    ])
      .then(([initiativeRes, verificacionRes]) => {
        setInitiative(initiativeRes.initiative);
        if (verificacionRes.yaPostulado) {
          setYaPostulado(true);
        }
      })
      .catch((requestError) => setError(requestError.message))
      .finally(() => setLoading(false));
  }, [token, initiativeId]);

  function updateForm({ target }) {
    setForm((current) => ({
      ...current,
      [target.name]: target.type === "checkbox" ? target.checked : target.value,
    }));
  }

  async function submitApplication(event) {
    event.preventDefault();
    if (
      !form.phone.trim() ||
      !form.motivation.trim() ||
      !form.acceptedTerms ||
      (!form.saturday && !form.sunday)
    )
      return showNotice({
        type: "error",
        text: "Completa tu teléfono, motivación, días disponibles y acepta los lineamientos.",
      });
    setBusy(true);
    try {
      await api.apply(token, initiative.id, form);
      setInitiative((current) => ({
        ...current,
        applicationStatus: "pending",
      }));
      setYaPostulado(true);
      setStep("success");
    } catch (requestError) {
      showNotice({ type: "error", text: requestError.message });
    } finally {
      setBusy(false);
    }
  }

  /** Cancela la iniciativa, conserva el registro histórico y avisa a sus participantes. */
  async function cancelInitiative() {
    setCancelling(true);
    try {
      await api.cancelInitiative(token, initiative.id);
      setInitiative((current) => ({ ...current, status: 'cancelled' }));
      setCancelOpen(false);
      showNotice({ text: 'La iniciativa fue cancelada y se conservará en el historial.' });
    } catch (requestError) {
      showNotice({ type: 'error', text: requestError.message });
    } finally {
      setCancelling(false);
    }
  }

  if (loading)
    return (
      <section className="workspace-content">
        <Loader />
      </section>
    );
  if (error || !initiative)
    return (
      <section className="workspace-content">
        <EmptyState
          title="No pudimos abrir esta iniciativa"
          text={error || "Vuelve a explorar las iniciativas disponibles."}
          action={() => go("/volunteer/explore")}
        />
      </section>
    );

  const canBoard =
    initiative.isMember ||
    initiative.applicationStatus === "accepted" ||
    (user.role === "company" && initiative.companyId === user.id);
  const available = Math.max(initiative.quota - initiative.joinedCount, 0);
  const isOwner = user.role === 'company' && initiative.companyId === user.id;
  const isCancelled = initiative.status === 'cancelled';

  if (step === "form")
    return (
      <ApplicationForm
        initiative={initiative}
        available={available}
        form={form}
        busy={busy}
        onChange={updateForm}
        onSubmit={submitApplication}
        onCancel={() => setStep("detail")}
      />
    );

  return (
    <section className="workspace-content initiative-detail">
      <button
        className="back-link"
        onClick={() =>
          go(
            user.role === "company"
              ? "/company/dashboard"
              : "/volunteer/explore",
          )
        }
      >
        <ArrowLeft size={18} /> Regresar
      </button>
      <article className="initiative-detail__card">
        <header>
          <div>
            {user.role === 'volunteer' ? (
              <button className="company-profile-link" onClick={() => go(`/company/profile/${initiative.companyId}`)}><Building2 size={15} /> Empresa: {initiative.companyName}</button>
            ) : <p className="company-label">Empresa: {initiative.companyName}</p>}
            <h1>{initiative.title}</h1>
          </div>
          <span className="category-chip">{initiative.category}</span>
        </header>
        <div className="initiative-detail__content">
          <div>
            <section>
              <h2>Información básica</h2>
              <p className="info-panel">{initiative.summary}</p>
              <div className="info-lines">
                <span>
                  <CalendarDays size={20} />
                  {initiative.date}
                </span>
                <span>
                  <MapPin size={20} />
                  {initiative.location}
                </span>
              </div>
            </section>
            <section>
              <h2>Requisitos</h2>
              <ul className="requirements">
                {initiative.requirements.map((requirement) => (
                  <li key={requirement}>
                    <CheckCircle2 size={17} />
                    {requirement}
                  </li>
                ))}
              </ul>
            </section>
          </div>
          <div>
            <div className={`detail-visual cover--${initiative.cover}`}>
              <div className="cover-art">
                <i />
                <i />
                <i />
              </div>
              <span className="detail-visual__caption">
                <Leaf size={28} /> Acciones que dejan huella
              </span>
            </div>
            <p className="detail-capacity">
              <Users size={16} /> {available} cupos disponibles
            </p>
          </div>
        </div>
        <footer>
          {isCancelled ? (
            <span className="initiative-cancelled"><XCircle size={17} /> Iniciativa cancelada</span>
          ) : canBoard ? (
            <Button
              onClick={() => go(`/kanban/${initiative.id}`)}
              icon={ArrowRight}
            >
              Ir al tablero Kanban
            </Button>
          ) : yaPostulado || initiative.applicationStatus === "pending" ? (
            // Si ya postuló (según la BD o localmente), mostramos el mensaje
            <Button disabled variant="secondary">
              <Clock3 size={17} /> Pendiente
            </Button>
          ) : user.role === "volunteer" ? (
            <Button onClick={() => setStep("form")} icon={HeartHandshake}>
              Postularse
            </Button>
          ) : null}
          {isOwner && !isCancelled && <Button variant="danger" onClick={() => setCancelOpen(true)} icon={XCircle}>Cancelar iniciativa</Button>}
        </footer>
      </article>
      {step === "success" && (
        <SuccessModal
          initiative={initiative}
          form={form}
          onClose={() => setStep("detail")}
        />
      )}
      {cancelOpen && <CancelInitiativeModal busy={cancelling} onCancel={() => setCancelOpen(false)} onConfirm={cancelInitiative} />}
    </section>
  );
}

function ApplicationForm({

  available,
  form,
  busy,
  onChange,
  onSubmit,
  onCancel,
}) {
  return (
    <section className="workspace-content application-page">
      <button className="back-link" onClick={onCancel}>
        <ArrowLeft size={18} /> Volver a iniciativa
      </button>
      <header className="application-page__head">
        <div>
          <span className="category-chip category-chip--soft">
            {initiative.category}
          </span>
          <h1>Postulación: {initiative.title}</h1>
          <p>
            Completa tus datos para enviar tu postulación a{" "}
            {initiative.companyName}.
          </p>
        </div>
        <span className="capacity-chip">
          <i /> {available} cupos restantes disponibles
        </span>
      </header>
      <div className="application-layout">
        <form className="application-form" onSubmit={onSubmit}>
          <FormSection
            number="1"
            title="Datos personales y de contacto"
            text="Información vinculada a tu cuenta para esta postulación."
          >
            <label className="field">
              <span>Nombre completo <em aria-hidden="true">*</em></span>
              <input
                name="fullName"
                value={form.fullName}
                onChange={onChange}
                required
              />
            </label>
            <div className="form-grid">
              <label className="field">
                <span>Correo electrónico <em aria-hidden="true">*</em></span>
                <input
                  type="email"
                  name="email"
                  value={form.email}
                  onChange={onChange}
                  required
                />
              </label>
              <label className="field">
                <span>Teléfono / WhatsApp <em aria-hidden="true">*</em></span>
                <input
                  name="phone"
                  value={form.phone}
                  onChange={onChange}
                  placeholder="Ej. +51 987 654 321"
                  required
                />
              </label>
            </div>
          </FormSection>

            title="Disponibilidad horaria"
            text="Selecciona el turno y los días en que te comprometes a participar."

                  key={option}
                >
                  <input
                    type="radio"
                    name="schedule"
                    value={option}
                    checked={form.schedule === option}
                    onChange={onChange}
                  />
                  <b>
                    Turno {option.startsWith("Mañana") ? "mañana" : "tarde"}
                  </b>
                  <span>{option}</span>
                </label>
              ))}
            </div>
            <p className="form-label">
              Días disponibles para la jornada inicial
            </p>
            <div className="day-options">
              <label>
                <input
                  type="checkbox"
                  name="saturday"
                  checked={form.saturday}
                  onChange={onChange}
                />{" "}
                Sábado
              </label>
              <label>
                <input
                  type="checkbox"
                  name="sunday"
                  checked={form.sunday}
                  onChange={onChange}
                />{" "}
                Domingo
              </label>
            </div>
          </FormSection>
          <FormSection
            number="3"
            title="Motivación y experiencia"
            text="Cuéntale a la organización tus motivaciones para unirte."
          >
            <label className="field">
              <span>¿Por qué te gustaría participar en esta iniciativa? <em aria-hidden="true">*</em></span>
              <textarea
                name="motivation"
                value={form.motivation}
                onChange={onChange}
                placeholder="Comparte brevemente tu motivación..."
                required
              />
            </label>
            <label className="field">
              <span>
                ¿Tienes experiencia previa en voluntariados similares?
                (Opcional)
              </span>
              <textarea
                name="experience"
                value={form.experience}
                onChange={onChange}
                placeholder="Menciona si participaste antes en actividades similares."
              />
            </label>
          </FormSection>
          <FormSection
            number="4"
            title="Logística y requerimientos"
            text="Esta información ayuda a preparar tu jornada de voluntariado."
          >
            <div className="form-grid">
              <label className="field">
                <span>Talla de camiseta / polo</span>
                <select
                  name="shirtSize"
                  value={form.shirtSize}
                  onChange={onChange}
                >
                  {["XS", "S", "M", "L", "XL"].map((size) => (
                    <option key={size}>{size}</option>
                  ))}
                </select>
              </label>
              <label className="field">
                <span>Restricción dietética o médica</span>
                <input
                  name="restrictions"
                  value={form.restrictions}
                  onChange={onChange}
                  placeholder="Ej. vegetariano, alergias..."
                />
              </label>
            </div>
            <label className="terms-check">
              <input
                type="checkbox"
                name="acceptedTerms"
                checked={form.acceptedTerms}
                onChange={onChange}
              />
              <span>
                <em aria-hidden="true">*</em> He leído y acepto los{" "}
                <b>lineamientos de participación y código de conducta</b>. Me
                comprometo a asistir puntualmente al turno confirmado.
              </span>
            </label>
          </FormSection>
          <footer className="application-form__footer">
            <Button type="button" variant="secondary" onClick={onCancel}>
              Cancelar
            </Button>
            <Button type="submit" icon={ArrowRight} disabled={busy}>
              {busy ? "Enviando..." : "Enviar postulación"}
            </Button>
          </footer>
        </form>
        <InitiativeSummary initiative={initiative} />
      </div>
    </section>
  );
}

        <div>
          <h2>{title}</h2>
          <p>{text}</p>
        </div>
      </header>
      {children}
    </section>
  );
}

function SuccessModal({ initiative, form, onClose }) {
  const days = [form.saturday && "Sábado", form.sunday && "Domingo"]
    .filter(Boolean)
    .join(" y ");
  return (
    <Modal onClose={onClose}>
      <div className="success-modal success-modal--application">
        <span className="success-icon">
          <CheckCircle2 size={34} />
        </span>
        <h2>¡Te has registrado correctamente!</h2>
        <p>
          Tu postulación a <b>{initiative.title}</b> ha sido recibida con éxito.
          Te notificaremos los próximos pasos.
        </p>
        <div className="success-details">
          <span>
            Turno seleccionado <b>{form.schedule}</b>
          </span>
          <span>
            Días <b>{days}</b>
          </span>
          <span>
            Organización <b>{initiative.companyName}</b>
          </span>
        </div>
        <div className="success-actions">
          <Button onClick={() => go("/volunteer/activities")} icon={ArrowRight}>
            Ir a mis actividades
          </Button>
          <Button variant="secondary" onClick={onClose}>
            Ver detalles
          </Button>
        </div>
      </div>
    </Modal>
  );
}

function CancelInitiativeModal({ busy, onCancel, onConfirm }) {
  return (
    <Modal onClose={onCancel}>
      <div className="confirm-modal">
        <span className="confirm-modal__icon"><XCircle size={31} /></span>
        <h2>¿Cancelar esta iniciativa?</h2>
        <p>La convocatoria dejará de recibir postulaciones y se conservará en el historial. Los voluntarios aceptados recibirán una notificación.</p>
        <div className="confirm-modal__actions">
          <Button variant="secondary" onClick={onCancel} disabled={busy}>Volver</Button>
          <Button variant="danger" onClick={onConfirm} disabled={busy}>{busy ? 'Cancelando...' : 'Sí, cancelar iniciativa'}</Button>
        </div>
      </div>
    </Modal>
  );
}
