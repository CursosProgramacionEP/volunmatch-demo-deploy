import { useEffect, useMemo, useState } from 'react';
import { CalendarDays, CheckCircle2, Clock3, Filter, MapPin, Search, Users, XCircle } from 'lucide-react';
import { Loader } from '../../components/comunes/Cargador.jsx';
import { EmptyState } from '../../components/comunes/EstadoVacio.jsx';
import { Button } from '../../components/comunes/Boton.jsx';
import { api } from '../../services/cliente-api.js';
import { go } from '../../navigation/navegacion.js';

const STATUS_LABELS = {
  active: 'Activa',
  completed: 'Completada',
  pending: 'Pendiente',
  accepted: 'Aceptada',
  rejected: 'Rechazada',
  cancelled: 'Cancelada',
};

/**
 * Consulta y filtra el historial de iniciativas de una empresa o voluntario.
 * @param {{token: string, user: object}} props Sesión autenticada para obtener el historial correspondiente.
 * @returns {JSX.Element} Vista de historial con filtros de búsqueda, estado y fechas.
 */
export function HistoryPage({ token, user }) {
  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [query, setQuery] = useState('');
  const [status, setStatus] = useState('all');
  const [from, setFrom] = useState('');
  const [to, setTo] = useState('');

  useEffect(() => {
    const request = user.role === 'company' ? api.myInitiatives(token) : api.activities(token);
    request
      .then((response) => setItems(user.role === 'company' ? response.initiatives : response.activities))
      .catch((requestError) => setError(requestError.message))
      .finally(() => setLoading(false));
  }, [token, user.role]);

  const records = useMemo(() => items.map((item) => ({ ...item, historyStatus: statusOf(item, user.role) })), [items, user.role]);
  const filtered = useMemo(() => records.filter((item) => {
    const matchesText = `${item.title} ${item.category} ${item.companyName || ''}`.toLowerCase().includes(query.toLowerCase().trim());
    const matchesStatus = status === 'all' || item.historyStatus === status;
    const date = new Date(item.date);
    const matchesFrom = !from || date >= new Date(`${from}T00:00:00`);
    const matchesTo = !to || date <= new Date(`${to}T23:59:59`);
    return matchesText && matchesStatus && matchesFrom && matchesTo;
  }), [records, query, status, from, to]);

  function clearFilters() {
    setQuery('');
    setStatus('all');
    setFrom('');
    setTo('');
  }

  const roleTitle = user.role === 'company' ? 'Historial de iniciativas' : 'Mi historial de voluntariado';
  const roleText = user.role === 'company'
    ? 'Consulta tus convocatorias activas, finalizadas o canceladas.'
    : 'Revisa tus postulaciones y actividades registradas.';

  return (
    <section className="workspace-content history-page">
      <div className="page-title"><div><p className="section-kicker">TRAYECTORIA Y SEGUIMIENTO</p><h1>{roleTitle}</h1><p>{roleText}</p></div><span className="page-title__badge"><Clock3 size={18} />{filtered.length} registros</span></div>
      <section className="history-filters">
        <label className="search-box"><Search size={20} /><input value={query} onChange={(event) => setQuery(event.target.value)} placeholder="Buscar iniciativas..." /></label>
        <label><span>Estado</span><select value={status} onChange={(event) => setStatus(event.target.value)}><option value="all">Todos los estados</option>{Object.entries(STATUS_LABELS).map(([value, label]) => <option key={value} value={value}>{label}</option>)}</select></label>
        <label><span>Desde</span><input type="date" value={from} onChange={(event) => setFrom(event.target.value)} /></label>
        <label><span>Hasta</span><input type="date" value={to} onChange={(event) => setTo(event.target.value)} /></label>
        <Button variant="secondary" icon={Filter} onClick={clearFilters}>Limpiar</Button>
      </section>
      {loading ? <Loader label="Cargando historial..." /> : error ? <EmptyState title="No pudimos cargar el historial" text={error} /> : filtered.length ? (
        <div className="history-list">{filtered.map((item) => <HistoryItem item={item} key={item.id} role={user.role} />)}</div>
      ) : <EmptyState title="No encontramos resultados" text="Ajusta los filtros o registra nuevas actividades." action={clearFilters} actionLabel="Limpiar filtros" />}
    </section>
  );
}

function HistoryItem({ item, role }) {
  const label = STATUS_LABELS[item.historyStatus] || 'Activa';
  return (
    <article className="history-item">
      <span className={`history-status history-status--${item.historyStatus}`}>{statusIcon(item.historyStatus)} {label}</span>
      <div className="history-item__main"><p className="company-label">{role === 'company' ? item.category : `Empresa: ${item.companyName}`}</p><h2>{item.title}</h2><p>{item.summary}</p><div><span><CalendarDays size={15} />{formatDate(item.date)}</span><span><MapPin size={15} />{item.location || 'Lugar por definir'}</span>{role === 'company' && <span><Users size={15} />Cupo: {item.quota}</span>}</div></div>
      <Button variant="secondary" onClick={() => go(`/initiative/${item.id}`)}>Ver detalles</Button>
    </article>
  );
}

function statusOf(item, role) {
  if (item.status === 'cancelled') return 'cancelled';
  if (role === 'volunteer') return item.applicationStatus || 'pending';
  if (item.status === 'done') return 'completed';
  return 'active';
}

function statusIcon(status) {
  if (status === 'completed' || status === 'accepted') return <CheckCircle2 size={14} />;
  if (status === 'cancelled' || status === 'rejected') return <XCircle size={14} />;
  return <Clock3 size={14} />;
}

function formatDate(value) {
  if (!value) return 'Sin fecha';
  return new Intl.DateTimeFormat('es-PE', { day: '2-digit', month: 'short', year: 'numeric' }).format(new Date(value));
}
