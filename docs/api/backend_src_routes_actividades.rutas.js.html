// Ruta de actividades, montada en /api/activities
//   GET /api/activities -> TODAS las iniciativas a las que el voluntario se postuló

import { Router } from 'express';
import pool from '../config/db.js';
import { auth } from '../middlewares/autenticacion.js';

export const activitiesRouter = Router();

/**
 * Adapta la iniciativa y el estado de postulación al formato que consume React.
 * @param {object} row Fila de iniciativa, empresa y postulación obtenida de PostgreSQL.
 * @param {Array<object>} [tasks=[]] Tareas relacionadas con la iniciativa.
 * @returns {object} Actividad lista para la interfaz del voluntario.
 */
function serializeInitiative(row, tasks = []) {
  const statusMap = {
    'pendiente': 'pending',
    'en_proceso': 'in_progress',
    'finalizado': 'done',
  };
  const initiativeStatusMap = {
    'activa': 'active',
    'finalizada': 'completed',
    'cancelada': 'cancelled',
  };

  // Mapear el estado de la postulación al frontend
  const applicationStatusMap = {
    'pendiente': 'pending',
    'aceptado': 'accepted',
    'rechazado': 'rejected',
  };

  return {
    id: row.idiniciativa,
    companyId: row.idempresa,
    companyName: row.nombreempresa || '',
    title: row.titulo,
    category: row.categoria,
    summary: row.descripcion,
    date: row.fechainicio,
    endDate: row.fechafin,
    location: row.lugar || '',
    quota: row.numcupo,
    joinedCount: 0,
    isMember: row.estado_postulacion === 'aceptado', // Solo es miembro si fue aceptado
    applicationStatus: applicationStatusMap[row.estado_postulacion] || 'pending',
    requirements: row.requisito ? row.requisito.split('\n').filter(Boolean) : [],
    cover: 'custom',
    members: [],
    tasks: tasks.map((t) => ({
  id: t.idtarea,
  title: t.titulo,
  description: t.descripcion,
  status: statusMap[t.estado] || 'pending',
  area: t.area || 'General',
  priority: t.priority || 'Media',
  schedule: 'Por definir',
  ownerId: t.idvoluntario_asignado || null,
  ownerName: t.voluntario_nombre || null,
  ownerEmail: t.voluntario_correo || null,
  ownerBio: t.voluntario_biografia || null,
  assignedAt: t.fecha_asignacion || null,
})),
    createdAt: row.creado_en || new Date().toISOString(),
    status: initiativeStatusMap[row.estado] || 'active',
  };
}

activitiesRouter.get('/', auth('volunteer'), async (req, res) => {
  try {
    // Traer TODAS las postulaciones del voluntario con su estado
    const result = await pool.query(
      `SELECT i.*, e.nombreempresa, p.estado AS estado_postulacion
       FROM postulacion p
       JOIN iniciativa i ON p.idiniciativa = i.idiniciativa
       JOIN empresa e ON i.idempresa = e.idempresa
       WHERE p.idvoluntario = $1
       ORDER BY p.fecha_postulacion DESC`,
      [req.user.id]
    );

    const ids = result.rows.map((r) => r.idiniciativa);
    let tasksByInitiative = {};
    if (ids.length) {
      const tasksResult = await pool.query(
        `SELECT t.*, v.nombrecompleto AS voluntario_nombre, v.correo AS voluntario_correo,
                v.biografia AS voluntario_biografia
         FROM tarea t
         LEFT JOIN voluntario v ON v.idvoluntario = t.idvoluntario_asignado
         WHERE t.idiniciativa = ANY($1)
         ORDER BY t.idtarea ASC`,
        [ids]
      );
      tasksResult.rows.forEach((t) => {
        if (!tasksByInitiative[t.idiniciativa]) tasksByInitiative[t.idiniciativa] = [];
        tasksByInitiative[t.idiniciativa].push(t);
      });
    }

    return res.json({
      activities: result.rows.map((row) =>
        serializeInitiative(row, tasksByInitiative[row.idiniciativa] || [])
      ),
    });
  } catch (err) {
    console.error('Error al listar actividades:', err.message);
    return res.status(500).json({ message: 'Error al cargar tus actividades.' });
  }
});
