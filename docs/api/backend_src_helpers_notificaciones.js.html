/**
 * Inserta una notificación persistente para una empresa o un voluntario.
 * @param {object} client Conexión de PostgreSQL o transacción activa.
 * @param {object} notification Datos visibles y destinatario de la notificación.
 * @returns {Promise<number|null>} Identificador creado o null si no hay destinatario.
 */
export async function createNotification(client, {
  volunteerId = null,
  companyId = null,
  type = 'system',
  title,
  message,
  destinationUrl = null,
}) {
  if (!volunteerId && !companyId) return null;

  const result = await client.query(
    `INSERT INTO notificacion
      (idvoluntario, idempresa, tipo, titulo, mensaje, url_destino)
     VALUES ($1, $2, $3, $4, $5, $6)
     RETURNING idnotificacion`,
    [volunteerId, companyId, type, title, message, destinationUrl],
  );

  return result.rows[0]?.idnotificacion ?? null;
}
