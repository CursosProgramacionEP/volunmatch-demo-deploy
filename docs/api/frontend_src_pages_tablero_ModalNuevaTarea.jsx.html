import { useState } from 'react';
import { Button } from '../../components/comunes/Boton.jsx';
import { Field } from '../../components/comunes/Campo.jsx';
import { Modal } from '../../components/comunes/VentanaModal.jsx';

/**
 * Reutiliza la ventana modal y los campos compartidos para crear una tarea Kanban.
 * @param {{onCrear: Function, onCancelar: Function}} props Acciones para persistir o cerrar el formulario.
 * @returns {JSX.Element} Formulario modal de tarea.
 */
export function ModalNuevaTarea({ onCrear, onCancelar }) {
  const [form, setForm] = useState({ title: '', area: 'General', description: '', priority: 'Media' });
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');

  function update(event) {
    const { name, value } = event.target;
    setForm((current) => ({ ...current, [name]: value }));
    setError('');
  }

  async function submit(event) {
    event.preventDefault();
    if (!form.title.trim()) {
      setError('El nombre de la tarea es obligatorio.');
      return;
    }
    setSaving(true);
    try {
      await onCrear(form);
    } catch (requestError) {
      setError(requestError.message || 'No se pudo crear la tarea.');
    } finally {
      setSaving(false);
    }
  }

  return (
    <Modal onClose={onCancelar}>
      <form className="shared-modal-form" onSubmit={submit} noValidate>
        <header><h2>Crear tarea</h2><p>Los voluntarios aceptados podrán tomarla y moverla en el tablero.</p></header>
        <div className="form-grid"><Field label="Nombre" name="title" value={form.title} onChange={update} error={error} placeholder="Nombre de la tarea" required autoFocus /><Field label="Área" name="area" value={form.area} onChange={update} placeholder="Ej. Logística" required /></div>
        <Field label="Descripción" name="description" as="textarea" value={form.description} onChange={update} placeholder="¿Qué debe hacer el voluntario?" />
        <Field label="Prioridad" name="priority" as="select" value={form.priority} onChange={update} options={['Baja', 'Media', 'Alta']} required />
        <footer className="form-footer"><Button type="button" variant="secondary" onClick={onCancelar}>Cancelar</Button><Button type="submit" disabled={saving}>{saving ? 'Creando...' : 'Crear tarea'}</Button></footer>
      </form>
    </Modal>
  );
}
