import { useEffect, useState } from 'react';
import { Bell, CheckCheck, Clock3 } from 'lucide-react';
import { Loader } from '../../components/comunes/Cargador.jsx';
import { EmptyState } from '../../components/comunes/EstadoVacio.jsx';
import { Button } from '../../components/comunes/Boton.jsx';
import { api } from '../../services/cliente-api.js';
import { go } from '../../navigation/navegacion.js';

/**
 * Centraliza las notificaciones persistentes del usuario autenticado.
 * @param {{token: string, showNotice: Function}} props Token de acceso y emisor de avisos locales.
 * @returns {JSX.Element} Bandeja de notificaciones con lectura masiva.
 */
export function NotificationsPage({ token, showNotice }) {
  const [notifications, setNotifications] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');

  useEffect(() => {
    api.notifications(token)
      .then(({ notifications: nextNotifications }) => setNotifications(nextNotifications))
      .catch((requestError) => setError(requestError.message))
      .finally(() => setLoading(false));
  }, [token]);

  async function markAllRead() {
    try {
      await api.markNotificationsRead(token);
      setNotifications((current) => current.map((notification) => ({ ...notification, read: true })));
      showNotice({ text: 'Todas las notificaciones fueron marcadas como leídas.' });
    } catch (requestError) {
      showNotice({ type: 'error', text: requestError.message });
    }
  }

  return (
    <section className="workspace-content notifications-page">
      <div className="page-title page-title--split"><div><p className="section-kicker">CENTRO DE AVISOS</p><h1>Bandeja de notificaciones</h1><p>Consulta postulaciones, tareas e información relevante para tu cuenta.</p></div><Button variant="secondary" icon={CheckCheck} onClick={markAllRead} disabled={!notifications.some((item) => !item.read)}>Marcar todas como leídas</Button></div>
      {loading ? <Loader label="Cargando notificaciones..." /> : error ? <EmptyState title="No pudimos cargar las notificaciones" text={error} /> : notifications.length ? <div className="notifications-list">{notifications.map((item) => <NotificationItem item={item} key={item.id} />)}</div> : <EmptyState title="No tienes notificaciones" text="Aquí aparecerán los avisos sobre tus iniciativas y actividades." />}
    </section>
  );
}

function NotificationItem({ item }) {
  return <button type="button" className={`notification-row ${item.read ? '' : 'notification-row--unread'}`} onClick={() => go(item.destinationUrl || '/notifications')}><span><Bell size={20} /></span><div><header><b>{item.title}</b>{!item.read && <i>No leída</i>}</header><p>{item.message}</p><small><Clock3 size={13} />{formatDate(item.createdAt)}</small></div></button>;
}

function formatDate(value) {
  if (!value) return 'Ahora';
  return new Intl.DateTimeFormat('es-PE', { dateStyle: 'medium', timeStyle: 'short' }).format(new Date(value));
}
